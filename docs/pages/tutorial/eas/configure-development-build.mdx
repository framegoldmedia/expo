// components/EarningsTracker.js (Updated)
import React, { useEffect, useState } from 'react';
import { useWallet } from '../context/WalletContext';

export default function EarningsTracker() {
  const { earnings } = useWallet();
  const [animatedIncome, setAnimatedIncome] = useState({ hourly: 0, daily: 0 });

  useEffect(() => {
    const interval = setInterval(() => {
      setAnimatedIncome(prev => ({
        hourly: prev.hourly + (earnings.hourly - prev.hourly) * 0.1,
        daily: prev.daily + (earnings.daily - prev.daily) * 0.1
      }));
    }, 300);
    
    return () => clearInterval(interval);
  }, [earnings]);

  return (
    <div className="bg-gray-900 p-4 rounded-lg gold-border">
      <h2 className="text-xl font-semibold gold-text">Earnings Tracker</h2>
      <p className="text-green-400 mt-2">
        Hourly: PGK {animatedIncome.hourly.toFixed(2)}
      </p>
      <p className="text-blue-400">
        Daily: PGK {animatedIncome.daily.toFixed(2)}
      </p>
    </div>
  );
}

// components/WithdrawForm.js (Enhanced)
import React, { useState } from 'react';
import { useWallet } from '../context/WalletContext';

export default function WithdrawForm() {
  const [formData, setFormData] = useState({
    bank: 'BSP',
    accountName: '',
    accountNumber: '',
    amount: ''
  });
  
  const { withdraw } = useWallet();

  const handleSubmit = async (e) => {
    e.preventDefault();
    await withdraw(formData);
  };

  return (
    <div className="bg-gray-900 p-4 rounded-lg gold-border">
      <h2 className="text-lg font-semibold gold-text mb-2">Withdraw Earnings</h2>
      <form onSubmit={handleSubmit}>
        <div className="mb-3">
          <label className="block mb-1 gold-text">Bank:</label>
          <select 
            value={formData.bank}
            onChange={(e) => setFormData({...formData, bank: e.target.value})}
            className="w-full p-2 bg-gray-800 rounded gold-border"
          >
            <option value="BSP">BSP</option>
            <option value="Kina">Kina Bank</option>
          </select>
        </div>
        
        <div className="mb-3">
          <input 
            placeholder="Account Name" 
            className="w-full p-2 bg-gray-800 rounded gold-border"
            value={formData.accountName}
            onChange={(e) => setFormData({...formData, accountName: e.target.value})}
          />
        </div>
        
        <div className="mb-3">
          <input 
            placeholder="Account Number"

  );
}

// context/WalletContext.js
import { createContext, useState, useEffect } from 'react';

export const WalletContext = createContext();

export function WalletProvider({ children }) {
  const [walletConnected, setWalletConnected] = useState(false);
  const [walletAddress, setWalletAddress] = useState('');
  const [earnings, setEarnings] = useState({ hourly: 0, daily: 0 });
  const [balance, setBalance] = useState(0);

  // Simulate earnings updates
  useEffect(() => {
    const interval = setInterval(() => {
      setEarnings(prev => ({
        hourly: prev.hourly + Math.random() * 10,
        daily: prev.daily + Math.random() * 50
      }));
    }, 5000);
    return () => clearInterval(interval);
  }, []);

  const connectWallet = async () => {
    // Actual Binance Chain wallet connection would go here
    setWalletConnected(true);
    setWalletAddress('0x71C7656EC7ab88b098defB751B7401B5f6d8976F');
  };

  const withdraw = async (withdrawalData) => {
    // Actual withdrawal logic
    console.log('Processing withdrawal:', withdrawalData);
    setBalance(prev => prev - parseFloat(withdrawalData.amount));
  };

  return (
    <WalletContext.Provider value={{
      walletConnected,
      walletAddress,
      connectWallet,
      earnings,
      balance,
      withdraw
    }}>
      {children}
    </WalletContext.Provider>
  );
}
